"""Trusted pytest collector, loaded only inside the sandbox."""
import json
from pathlib import Path
from typing import Any, Generator
import pytest

phases: list[dict[str, Any]] = []
collected: list[str] = []
collection_errors: list[str] = []
collection_complete = False

def pytest_collection_finish(session: Any) -> None:
    global collection_complete
    collected.extend(item.nodeid for item in session.items)
    collection_complete = True

def pytest_collectreport(report: Any) -> None:
    if report.failed:
        collection_errors.append(str(report.longrepr)[:8192])

@pytest.hookimpl(hookwrapper=True)
def pytest_runtest_makereport(item: Any, call: Any) -> Generator[None, Any, None]:
    outcome = yield
    report = outcome.get_result()
    exception_type = message = None
    truncated = False
    if call.excinfo is not None:
        kind = call.excinfo.type
        exception_type = kind.__name__ if kind.__module__ == 'builtins' else kind.__module__ + '.' + kind.__qualname__
        raw_message = str(call.excinfo.value)
        truncated = len(raw_message) > 4000
        message = raw_message[:4000]
    phases.append({
        'node_id': report.nodeid, 'phase': report.when, 'outcome': report.outcome,
        'was_xfail': hasattr(report, 'wasxfail'), 'exception_type': exception_type,
        'message': message, 'traceback': str(report.longrepr)[:8192] if report.failed else None,
        'truncated': truncated or (report.failed and len(str(report.longrepr)) > 8192),
    })

@pytest.hookimpl(trylast=True)
def pytest_sessionfinish(session: Any, exitstatus: Any) -> None:
    payload = {
        'schema_version': '1.0', 'pytest_version': pytest.__version__, 'collected': collected,
        'collection_complete': collection_complete, 'collection_errors': collection_errors,
        'phases': phases, 'session_complete': True, 'exit_code': int(exitstatus),
    }
    Path('/tmp/repo-repro-pytest.json').write_text(json.dumps(payload), encoding='utf-8')
